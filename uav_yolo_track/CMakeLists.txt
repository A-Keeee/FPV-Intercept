cmake_minimum_required(VERSION 3.16)
project(uav_yolo_track)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(uav_common_msg REQUIRED)
find_package(uav_ard_mav REQUIRED)
find_package(Threads REQUIRED)

find_program(ULTRALYTICS_PYTHON_EXECUTABLE NAMES python3 python
  HINTS "$ENV{HOME}/anaconda3/envs/yolo/bin" "$ENV{HOME}/miniconda3/envs/yolo/bin"
  DOC "Interpreter with Ultralytics, torch, numpy, PyYAML, OpenCV, lap and scipy installed" REQUIRED)
message(STATUS "Native Ultralytics tracker interpreter: ${ULTRALYTICS_PYTHON_EXECUTABLE}")

add_library(yolo_track_core STATIC src/native_tracker.cpp src/pipeline.cpp)
target_include_directories(yolo_track_core PUBLIC include)
target_compile_definitions(yolo_track_core PRIVATE UAV_YOLO_PYTHON="${ULTRALYTICS_PYTHON_EXECUTABLE}")
target_link_libraries(yolo_track_core PUBLIC uav_ard_mav::ard_mav_inference Threads::Threads)

add_executable(uav_yolo_track src/node.cpp)
target_link_libraries(uav_yolo_track yolo_track_core)
ament_target_dependencies(uav_yolo_track ament_index_cpp rclcpp sensor_msgs std_msgs cv_bridge uav_common_msg)
add_executable(uav_yolo_single_track src/node.cpp)
target_compile_definitions(uav_yolo_single_track PRIVATE UAV_YOLO_SINGLE_MODEL=1)
target_link_libraries(uav_yolo_single_track yolo_track_core)
ament_target_dependencies(uav_yolo_single_track ament_index_cpp rclcpp sensor_msgs std_msgs cv_bridge uav_common_msg)
add_executable(yolo_track_video src/video.cpp)
target_link_libraries(yolo_track_video yolo_track_core)
ament_target_dependencies(yolo_track_video ament_index_cpp)
add_executable(yolo_single_track_video src/video.cpp)
target_compile_definitions(yolo_single_track_video PRIVATE UAV_YOLO_SINGLE_MODEL=1)
target_link_libraries(yolo_single_track_video yolo_track_core)
ament_target_dependencies(yolo_single_track_video ament_index_cpp)

install(TARGETS uav_yolo_track uav_yolo_single_track yolo_track_video yolo_single_track_video DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_native_tracker test/test_native_tracker.cpp)
  target_link_libraries(test_native_tracker yolo_track_core)
  ament_target_dependencies(test_native_tracker ament_index_cpp)
  target_compile_definitions(test_native_tracker PRIVATE TRACKER_SHARE="${CMAKE_CURRENT_SOURCE_DIR}")
endif()
ament_package()
